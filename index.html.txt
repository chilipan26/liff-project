<div id="createArea" class="card hidden">
  <h3>新規案件登録</h3>
  <input id="workDate" type="date" placeholder="勤務日">
  <input id="projectName" placeholder="案件名">
  <input id="placeName" placeholder="現場名">
  <input id="address" placeholder="現場住所">
  <input id="meetTime" type="time">
  <input id="startTime" type="time">
  <input id="endTime" type="time">
  <input id="people" type="number" placeholder="人数">
  <input id="client" placeholder="請求先">
  <input id="billingGroup" placeholder="請求グループ">
  <input id="priceType" placeholder="単価区分">
  <input id="manager" placeholder="担当者">
  <textarea id="memo" placeholder="備考"></textarea>
  <button onclick="createProject()">登録する</button>
</div>

<div id="searchArea" class="card hidden">
  <h3>案件検索・訂正</h3>
  <input id="keyword" placeholder="案件名・日付・現場名で検索">
  <button onclick="searchProject()">検索</button>
  <div id="results"></div>
</div>

<script>
const GAS_URL = "ここにGASのWebアプリURLを貼る";

function showCreate() {
  document.getElementById("createArea").classList.remove("hidden");
  document.getElementById("searchArea").classList.add("hidden");
}

function showSearch() {
  document.getElementById("searchArea").classList.remove("hidden");
  document.getElementById("createArea").classList.add("hidden");
}

async function createProject() {
  const data = {
    action: "create",
    workDate: workDate.value,
    projectName: projectName.value,
    placeName: placeName.value,
    address: address.value,
    meetTime: meetTime.value,
    startTime: startTime.value,
    endTime: endTime.value,
    people: people.value,
    client: client.value,
    billingGroup: billingGroup.value,
    priceType: priceType.value,
    manager: manager.value,
    memo: memo.value,
    updatedBy: "管理者"
  };

  const res = await fetch(GAS_URL, {
    method: "POST",
    body: JSON.stringify(data)
  });

  const json = await res.json();
  alert("登録しました：" + json.projectId);
}

async function searchProject() {
  const res = await fetch(GAS_URL, {
    method: "POST",
    body: JSON.stringify({
      action: "search",
      keyword: keyword.value
    })
  });

  const json = await res.json();
  const box = document.getElementById("results");
  box.innerHTML = "";

  json.results.forEach(item => {
    box.innerHTML += `
      <div class="card">
        <b>${item["案件名"] || ""}</b><br>
        日付：${item["勤務日"] || ""}<br>
        現場：${item["現場名"] || ""}<br>
        請求先：${item["請求先"] || ""}<br>
        <button onclick='openEdit(${JSON.stringify(item)})'>訂正する</button>
      </div>
    `;
  });
}

function openEdit(item) {
  const newName = prompt("案件名を訂正", item["案件名"] || "");
  if (newName === null) return;

  updateProject(item["案件ID"], {
    "案件名": newName
  });
}

async function updateProject(projectId, updates) {
  await fetch(GAS_URL, {
    method: "POST",
    body: JSON.stringify({
      action: "update",
      projectId,
      updates,
      reason: "LIFFから訂正",
      updatedBy: "管理者"
    })
  });

  alert("訂正しました");
}
